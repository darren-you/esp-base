cmake_minimum_required(VERSION 3.16)
if(NOT IDF_TARGET)
    if(NOT "$ENV{IDF_TARGET}" STREQUAL "")
        set(IDF_TARGET "$ENV{IDF_TARGET}")
    else()
        set(IDF_TARGET "esp32c3")
    endif()
endif()
if(IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR
        "ESP32 Base image is blocked until P6-03 freezes its own partition table, OTA policy and signed boot chain")
elseif(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "ESP Base supports only esp32c3 and esp32 targets")
endif()
set(ESP_BASE_APP "esp_base" CACHE STRING "esp_base or mqtt_integration")
if(ESP_BASE_APP STREQUAL "esp_base")
    set(PROJECT_VER "0.2.0")
    set(ESP_BASE_PROJECT "esp_base")
elseif(ESP_BASE_APP STREQUAL "mqtt_integration")
    set(PROJECT_VER "0.3.0-lab")
    set(ESP_BASE_PROJECT "esp_base_mqtt_lab")
    if(NOT DEFINED SDKCONFIG_DEFAULTS)
        if(DEFINED ENV{SDKCONFIG_DEFAULTS})
            set(SDKCONFIG_DEFAULTS "$ENV{SDKCONFIG_DEFAULTS}")
        else()
            set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
        endif()
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/apps/mqtt_integration/sdkconfig.defaults")
else()
    message(FATAL_ERROR "Unknown ESP_BASE_APP")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/apps/${ESP_BASE_APP}/main")
if(ESP_BASE_CONTAINER_BINDING_PROBE)
    set(WAMR_BUILD_INSTRUCTION_METERING 1 CACHE BOOL "Bound guest instructions" FORCE)
    set(WAMR_BUILD_BULK_MEMORY 0 CACHE BOOL "Avoid implicit constructor invocation" FORCE)
    set(WAMR_BUILD_SHARED_MEMORY 0 CACHE BOOL "Disallow guest shared memory" FORCE)
    set(WAMR_BUILD_SHRUNK_MEMORY 0 CACHE BOOL "Preserve standard Wasm page sizes" FORCE)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/integrations/container_binding")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${ESP_BASE_PROJECT})
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../tools/check_sdk.py"
    --path "${IDF_PATH}"
    RESULT_VARIABLE sdk_result ERROR_VARIABLE sdk_error)
if(NOT sdk_result STREQUAL "0")
    message(FATAL_ERROR "${sdk_error}")
endif()
idf_component_get_property(lwip_component lwip COMPONENT_DIR)
get_filename_component(actual_lwip_component "${lwip_component}" REALPATH)
get_filename_component(expected_lwip_component "${IDF_PATH}/components/lwip" REALPATH)
if(NOT actual_lwip_component STREQUAL expected_lwip_component)
    message(FATAL_ERROR "ESP Base requires the locked SDK lwIP component")
endif()
if(ESP_BASE_APP STREQUAL "esp_base" AND (CONFIG_EMQTT_PLAINTEXT_LAB OR ESP_BASE_MQTT_LAB_INPUTS))
    message(FATAL_ERROR "Normal ESP Base build rejects MQTT lab inputs and plaintext test mode")
endif()
if(ESP_BASE_APP STREQUAL "esp_base" AND (NOT CONFIG_MBEDTLS_MD5_C OR NOT CONFIG_LWIP_SO_LINGER OR
    NOT CONFIG_MBEDTLS_HAVE_TIME_DATE OR CONFIG_LWIP_MAX_SOCKETS LESS 12))
    message(FATAL_ERROR "ESP Base FRP requires MD5 token auth, SO_LINGER, trusted TLS time and 12 sockets")
endif()
