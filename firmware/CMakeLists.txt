cmake_minimum_required(VERSION 3.16)
set(ESP_BASE_APP "esp_base" CACHE STRING "esp_base or mqtt_integration")
if(ESP_BASE_APP STREQUAL "esp_base")
    set(PROJECT_VER "0.2.0")
    set(ESP_BASE_PROJECT "esp_base")
elseif(ESP_BASE_APP STREQUAL "mqtt_integration")
    set(PROJECT_VER "0.3.0-lab")
    set(ESP_BASE_PROJECT "esp_base_mqtt_lab")
    if(NOT DEFINED SDKCONFIG_DEFAULTS)
        if(DEFINED ENV{SDKCONFIG_DEFAULTS})
            set(SDKCONFIG_DEFAULTS "$ENV{SDKCONFIG_DEFAULTS}")
        else()
            set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
        endif()
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/apps/mqtt_integration/sdkconfig.defaults")
else()
    message(FATAL_ERROR "Unknown ESP_BASE_APP")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/apps/${ESP_BASE_APP}/main")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${ESP_BASE_PROJECT})
if(ESP_BASE_APP STREQUAL "esp_base" AND (CONFIG_EMQTT_PLAINTEXT_LAB OR ESP_BASE_MQTT_LAB_INPUTS))
    message(FATAL_ERROR "Normal ESP Base build rejects MQTT lab inputs and plaintext test mode")
endif()
