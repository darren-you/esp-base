cmake_minimum_required(VERSION 3.16)
if(NOT IDF_TARGET)
    set(IDF_TARGET esp32c3)
endif()
if(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "NVS capacity probe is scoped to ESP32-C3")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../components/remote_config")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(nvs_capacity_probe)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../../tools/check_sdk.py"
    --path "${IDF_PATH}" RESULT_VARIABLE sdk_result ERROR_VARIABLE sdk_error)
if(NOT sdk_result STREQUAL "0")
    message(FATAL_ERROR "${sdk_error}")
endif()
