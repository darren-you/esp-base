cmake_minimum_required(VERSION 3.16)
if(NOT IDF_TARGET)
    set(IDF_TARGET esp32c3)
endif()
if(NOT IDF_TARGET STREQUAL "esp32c3" AND NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "NVS capacity probe supports esp32c3 and esp32")
endif()
if(NOT DEFINED PROBE_NVS_PAGES OR NOT PROBE_NVS_PAGES MATCHES "^(6|8)$")
    message(FATAL_ERROR "Pass -D PROBE_NVS_PAGES=6 or 8")
endif()
if(IDF_TARGET STREQUAL "esp32" AND NOT PROBE_NVS_PAGES STREQUAL "6")
    message(FATAL_ERROR "ESP32 candidate base_store is exactly six pages")
endif()
# The resolved Container component uses the same WAMR profile as the product.
set(WAMR_BUILD_INSTRUCTION_METERING 1 CACHE BOOL "Bound guest instructions" FORCE)
set(WAMR_BUILD_BULK_MEMORY 0 CACHE BOOL "Avoid implicit constructor invocation" FORCE)
set(WAMR_BUILD_SHARED_MEMORY 0 CACHE BOOL "Disallow guest shared memory" FORCE)
set(WAMR_BUILD_SHRUNK_MEMORY 0 CACHE BOOL "Preserve standard Wasm page sizes" FORCE)
if(IDF_TARGET STREQUAL "esp32")
    set(PROBE_NVS_OFFSET 0x3fa000)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.esp32;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.esp32.6page")
else()
    set(PROBE_NVS_OFFSET 0x138000)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.${PROBE_NVS_PAGES}page")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../components/remote_config")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(IDF_TARGET STREQUAL "esp32")
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/dependencies.lock.esp32")
endif()
project(nvs_capacity_probe)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../../tools/check_sdk.py"
    --path "${IDF_PATH}" RESULT_VARIABLE sdk_result ERROR_VARIABLE sdk_error)
if(NOT sdk_result STREQUAL "0")
    message(FATAL_ERROR "${sdk_error}")
endif()
