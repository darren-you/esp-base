# host 测试：宿主核心（components/base_wasm/core）加 WAMR 的本机构建，用真实的应用（tests/apps/*.wasm）跑。
# WAMR 的特性开关要与设备一致：见 firmware/CMakeLists.txt 和 sdkconfig.defaults 的 WASM 一节。
cmake_minimum_required(VERSION 3.22)
project(wasm_host_test C)

set(WAMR_ROOT_DIR "" CACHE PATH "组件管理器取回的 WAMR：firmware/managed_components/wasm-micro-runtime")
set(BASE_WASM_DIR "${CMAKE_CURRENT_LIST_DIR}/../../components/base_wasm")

string(TOLOWER ${CMAKE_HOST_SYSTEM_NAME} WAMR_BUILD_PLATFORM)
if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(WAMR_BUILD_TARGET "AARCH64")
else()
    set(WAMR_BUILD_TARGET "X86_64")
endif()
set(CMAKE_BUILD_TYPE Release)
set(WAMR_BUILD_INTERP 1)
if(NOT DEFINED WAMR_BUILD_FAST_INTERP)
    set(WAMR_BUILD_FAST_INTERP 0)
endif()
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_BULK_MEMORY 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
# 设备上没有硬件越界保护：本机也用软件检查，行为才一致。
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_STACK_HW_BOUND_CHECK 1)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(vmlib STATIC ${WAMR_RUNTIME_LIB_SOURCE})

add_executable(wasm_host_test wasm_host_test.c ${BASE_WASM_DIR}/core/wasm_host.c)
target_include_directories(wasm_host_test PRIVATE ${BASE_WASM_DIR}/core)
target_compile_options(wasm_host_test PRIVATE -std=c11 -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(wasm_host_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(wasm_host_test vmlib m pthread)
